// --- Image Compression Utility ---

const compressImage = (base64Str, maxWidth = 800, quality = 0.7) => {

  return new Promise((resolve) => {

    const img = new window.Image();

    img.src = base64Str;

    img.onload = () => {

      const canvas = document.createElement('canvas');

      let width = img.width;

      let height = img.height;


      if (width > maxWidth) {

        height *= maxWidth / width;

        width = maxWidth;

      }


      canvas.width = width;

      canvas.height = height;

      const ctx = canvas.getContext('2d');

      ctx.drawImage(img, 0, 0, width, height);

      resolve(canvas.toDataURL('image/jpeg', quality));

    };

  });

};

  // Navigation Handlers for Lightbox

  const handleNextPreview = (e) => {

    e.stopPropagation();

    if (generatedImages.length <= 1) return;

    const currentIndex = generatedImages.findIndex(img => img.styleId === previewImage.styleId);

    const nextIndex = (currentIndex + 1) % generatedImages.length;

    setPreviewImage(generatedImages[nextIndex]);

  };


  const handlePrevPreview = (e) => {

    e.stopPropagation();

    if (generatedImages.length <= 1) return;

    const currentIndex = generatedImages.findIndex(img => img.styleId === previewImage.styleId);

    const prevIndex = (currentIndex - 1 + generatedImages.length) % generatedImages.length;

    setPreviewImage(generatedImages[prevIndex]);

  };


  useEffect(() => {

    const handleKeyDown = (e) => {

      if (!previewImage) return;

      if (e.key === 'ArrowRight') {

        const currentIndex = generatedImages.findIndex(img => img.styleId === previewImage.styleId);

        const nextIndex = (currentIndex + 1) % generatedImages.length;

        setPreviewImage(generatedImages[nextIndex]);

      } else if (e.key === 'ArrowLeft') {

        const currentIndex = generatedImages.findIndex(img => img.styleId === previewImage.styleId);

        const prevIndex = (currentIndex - 1 + generatedImages.length) % generatedImages.length;

        setPreviewImage(generatedImages[prevIndex]);

      } else if (e.key === 'Escape') {

        setPreviewImage(null);

      }

    };

    window.addEventListener('keydown', handleKeyDown);

    return () => window.removeEventListener('keydown', handleKeyDown);

  }, [previewImage, generatedImages]);



  // Gemini API Logic

  const callGeminiAPI = async (modelId, prompt, base64Data, mimeType) => {

    const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/${modelId}:generateContent`;

    const payload = {

        contents: [{

            parts: [

                { text: prompt },

                { inlineData: { mimeType: mimeType, data: base64Data } }

            ]

        }],

        generationConfig: {

            responseModalities: ["TEXT", "IMAGE"],

            temperature: 0.5, 

            candidateCount: 1,

        },

        safetySettings: [

            { category: "HARM_CATEGORY_HARASSMENT", threshold: "BLOCK_NONE" },

            { category: "HARM_CATEGORY_HATE_SPEECH", threshold: "BLOCK_NONE" },

            { category: "HARM_CATEGORY_SEXUALLY_EXPLICIT", threshold: "BLOCK_NONE" },

            { category: "HARM_CATEGORY_DANGEROUS_CONTENT", threshold: "BLOCK_NONE" }

        ]

    };


    let fetchUrl = apiUrl;

    if (typeof window !== 'undefined' && window.google_api_key) {

        fetchUrl += `?key=${window.google_api_key}`;

    }


    const response = await fetch(fetchUrl, {

        method: 'POST',

        headers: { 'Content-Type': 'application/json' },

        body: JSON.stringify(payload)

    });


    if (!response.ok) throw new Error(`API Error: ${response.status}`);

    const result = await response.json();

    const imagePart = result.candidates?.[0]?.content?.parts?.find(p => p.inlineData);

    if (imagePart) return `data:image/png;base64,${imagePart.inlineData.data}`;

    throw new Error("No image data in response");

  };


  const generateImages = async () => {

    if (!uploadedImage) { setErrorMsg('변환할 이미지를 먼저 업로드해주세요.'); return; }

    if (selectedStyles.length === 0) { setErrorMsg('최소 1개 이상의 스타일을 선택해주세요.'); return; }


    setIsGenerating(true);

    setErrorMsg('');

    setGeneratedImages([]);

    const base64Data = uploadedImage.split(',')[1];

    

    try {

      const promises = selectedStyles.map(async (styleId) => {

        const styleData = PRESET_STYLES.find(s => s.id === styleId);

        const modelToUse = 'gemini-2.5-flash-image-preview'; 

        const orientationText = imageOrientation === 'landscape' ? 'horizontal landscape (A6 aspect ratio)' : 'vertical portrait (A6 aspect ratio)';

        const finalPrompt = `TRANSFORM THIS IMAGE INTO THE ART STYLE OF: ${styleData.prompt}. CRITICAL: Apply the style EXTREMELY INTENSELY. Keep the main subject recognizable but REDRAW everything else. IMPORTANT: Zoom in or crop if needed for better composition. Output high-quality ${orientationText}. ${styleData.id === 'inkwash' ? 'STRICTLY BLACK AND WHITE.' : ''}`;


        try {

            const imageData = await callGeminiAPI(modelToUse, finalPrompt, base64Data, imageMimeType);

            return { styleId: styleData.id, styleName: styleData.name, imageData, orientation: imageOrientation };

        } catch (err) {

            console.error(`Model generation failed for ${styleData.name}:`, err);

            return null;

        }

      });


      const results = await Promise.all(promises);

      const successfulImages = results.filter(img => img !== null);

      

      if (successfulImages.length === 0) setErrorMsg('이미지 생성에 실패했습니다.');

      else {

        setGeneratedImages(successfulImages);

        const newHistoryItem = { id: Date.now(), date: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }), images: successfulImages, thumbnail: successfulImages[0].imageData };

        setHistory(prev => [newHistoryItem, ...prev]);

      }

    } catch (e) { setErrorMsg('오류가 발생했습니다.'); } 

    finally { setIsGenerating(false); }

  };


  // --- Mobile View Rendering ---

  if (isMobileMode) {

    return <MobileUploadView sessionId={qrSessionId} fb={fb} user={user} onClose={() => {}} />;

  }


  // --- PC View Rendering ---

  return (

    <div className="min-h-screen bg-gray-50 text-slate-800 font-sans selection:bg-indigo-100 pb-20 relative">

      <header className="bg-white shadow-sm border-b border-gray-100 sticky top-0 z-20">

        <div className="max-w-5xl mx-auto px-4 h-16 flex items-center justify-between">

          <div className="flex items-center gap-2">

            <div className="bg-indigo-600 text-white p-1.5 rounded-lg"><Sparkles size={20} /></div>

            <h1 className="text-xl font-bold bg-clip-text text-transparent bg-gradient-to-r from-indigo-600 to-purple-600">AI Style Studio <span className="text-xs font-medium text-gray-400 border border-gray-200 rounded px-1.5 py-0.5 ml-1">A6 Edition</span></h1>

          </div>

          <button onClick={handleReset} className="flex items-center gap-2 px-4 py-2 rounded-full text-sm font-medium text-gray-600 hover:bg-gray-100 hover:text-red-500 transition-colors"><RotateCcw size={16} /><span className="hidden sm:inline">전체 초기화</span></button>

        </div>

        {/* History Bar */}

        {history.length > 0 && (

          <div className="bg-gray-50 border-b border-gray-200">

            <div className="max-w-5xl mx-auto px-4 py-3 flex items-center gap-4">

                <div className="flex items-center gap-2 text-xs font-bold text-gray-500 whitespace-nowrap"><Clock size={14} /> 이전 작업</div>

                <div className="flex-1 overflow-x-auto whitespace-nowrap scrollbar-hide flex gap-3 py-1">

                    {history.map(item => (

                        <button key={item.id} onClick={() => loadHistoryItem(item)} className="group relative w-12 h-12 flex-shrink-0 rounded-lg overflow-hidden border-2 border-white shadow-sm hover:border-indigo-500 hover:scale-105 transition-all focus:outline-none focus:ring-2 focus:ring-indigo-500">

                            <img src={item.thumbnail} alt="History" className="w-full h-full object-cover" />

                            <div className="absolute inset-0 bg-black/0 group-hover:bg-black/10 transition-colors"></div>

                        </button>

                    ))}

                </div>

                <button onClick={handleClearHistory} className="p-1.5 text-gray-400 hover:text-red-500 rounded-full hover:bg-red-50 transition-colors"><Trash2 size={14} /></button>

            </div>

          </div>

        )}

      </header>


      <main className="max-w-5xl mx-auto px-4 py-8 space-y-8">

        {/* QR Modal */}

        {showQR && (

            <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 backdrop-blur-sm animate-in fade-in">

                <div className="bg-white p-8 rounded-2xl shadow-2xl max-w-sm w-full text-center relative">

                    <button onClick={() => setShowQR(false)} className="absolute top-4 right-4 text-gray-400 hover:text-gray-600"><X size={24} /></button>

                    <h3 className="text-xl font-bold text-gray-800 mb-4">스마트폰으로 스캔하세요</h3>

                    <div className="bg-white p-2 border-2 border-dashed border-indigo-200 rounded-xl inline-block mb-4">

                        <img 

                            src={`https://api.qrserver.com/v1/create-qr-code/?size=200x200&data=${encodeURIComponent(window.location.href + '?session=' + qrSessionId)}`} 

                            alt="Scan QR" 

                            className="w-48 h-48"

                        />

                    </div>

                    

                    <div className="bg-orange-50 border border-orange-200 p-4 rounded-xl text-xs text-left text-orange-800 mb-4">

                        <strong>⚠️ 테스트 팁:</strong><br/>

                        현재 미리보기 환경에서는 실제 스마트폰 접속이 안 될 수 있습니다. (정식 배포 시 정상 작동)<br/>

                        테스트를 위해 아래 버튼을 눌러보세요.

                    </div>


                    <button 

                        onClick={() => setShowSimulator(true)}

                        className="w-full py-3 bg-indigo-600 hover:bg-indigo-700 text-white rounded-xl font-bold flex items-center justify-center gap-2 transition-colors shadow-md"

                    >

                        <Smartphone size={18} />

                        가상 스마트폰 열기 (테스트)

                    </button>

                </div>

            </div>

        )}


        {/* Simulator Overlay */}

        {showSimulator && (

            <div className="fixed bottom-8 right-8 z-[60] w-[320px] h-[640px] bg-black rounded-[40px] border-8 border-black shadow-2xl overflow-hidden ring-4 ring-gray-200 animate-in slide-in-from-bottom-10 fade-in duration-300">

                <MobileUploadView sessionId={qrSessionId} fb={fb} user={user} onClose={() => setShowSimulator(false)} />

            </div>

        )}


        {/* Upload Section */}

        <section className="bg-white rounded-2xl p-8 shadow-sm border border-gray-100 text-center">

          {!uploadedImage ? (

            <div className="flex flex-col gap-4">

                <div onClick={() => fileInputRef.current?.click()} className="border-2 border-dashed border-gray-300 rounded-xl p-12 cursor-pointer hover:border-indigo-500 hover:bg-indigo-50/50 transition-all group">

                    <div className="w-16 h-16 bg-indigo-50 text-indigo-500 rounded-full flex items-center justify-center mx-auto mb-4 group-hover:scale-110 transition-transform"><Upload size={32} /></div>

                    <h3 className="text-lg font-semibold text-gray-700 mb-2">PC에서 사진 업로드</h3>

                    <p className="text-sm text-gray-500">클릭, 드래그, 또는 붙여넣기 (Ctrl+V)</p>

                </div>

                <div className="flex items-center gap-4 before:h-px before:flex-1 before:bg-gray-200 after:h-px after:flex-1 after:bg-gray-200 text-sm text-gray-400">OR</div>

                <button 

                    onClick={openQRModal}

                    className="w-full py-4 border border-gray-300 hover:border-indigo-500 hover:text-indigo-600 rounded-xl flex items-center justify-center gap-2 text-gray-600 font-medium transition-all"

                >

                    <QrCode size={20} />

                    스마트폰 QR 코드로 업로드

                </button>

            </div>

          ) : (

            <div className="flex flex-col items-center gap-4">

               <div className="relative group">

                  <img src={uploadedImage} alt="Original" className="max-h-64 rounded-lg shadow-md object-contain"/>

                  <button onClick={() => { setUploadedImage(null); setGeneratedImages([]); }} className="absolute -top-2 -right-2 bg-red-500 text-white p-1.5 rounded-full shadow-lg hover:bg-red-600 transition-colors"><X size={16} /></button>

               </div>

               <p className="text-sm font-medium text-gray-600 bg-gray-100 px-3 py-1 rounded-full">원본 이미지 ({imageMimeType})</p>

               <div className="flex items-center gap-2 px-3 py-1.5 rounded-lg text-xs animate-in fade-in bg-indigo-50 text-indigo-700">

                    <Zap size={14} className="fill-current" /> 

                    적용 엔진: <strong>Nano Banana (Style Maximized)</strong>

                    <span className="mx-1 text-gray-300">|</span>

                    <span>{imageOrientation === 'landscape' ? '가로형' : '세로형'} A6 출력</span>

               </div>

            </div>

          )}

          <input type="file" ref={fileInputRef} onChange={handleFileChange} accept="image/*" className="hidden" />

        </section>


        {/* Style Selection */}

        <section>

          <div className="flex items-center justify-between mb-4">

            <h2 className="text-lg font-bold text-gray-800 flex items-center gap-2"><Palette size={20} className="text-indigo-600"/> 스타일 선택</h2>

            <span className={`text-sm font-medium px-2 py-1 rounded-md ${selectedStyles.length === 4 ? 'bg-green-100 text-green-700' : 'bg-gray-100 text-gray-500'}`}>{selectedStyles.length} / 4 선택됨</span>

          </div>

          <div className="grid grid-cols-2 md:grid-cols-4 gap-3">

            {PRESET_STYLES.map((style) => {

              const isSelected = selectedStyles.includes(style.id);

              const isDisabled = !isSelected && selectedStyles.length >= 4;

              return (

                <button key={style.id} onClick={() => toggleStyle(style.id)} disabled={isDisabled}

                  className={`relative p-4 rounded-xl text-left transition-all duration-200 border ${isSelected ? 'border-indigo-600 bg-indigo-50 shadow-md ring-1 ring-indigo-600' : 'border-gray-200 bg-white hover:border-gray-300 hover:shadow-sm'} ${isDisabled ? 'opacity-50 cursor-not-allowed grayscale' : 'cursor-pointer'}`}>

                  <div className="flex justify-between items-start mb-2">

                    <span className={`text-sm font-bold ${isSelected ? 'text-indigo-700' : 'text-gray-700'}`}>{style.name.split(' (')[0]}</span>

                    {isSelected && <Check size={16} className="text-indigo-600" />}

                  </div>

                  <p className={`text-[10px] line-clamp-2 ${isSelected ? 'text-indigo-600/80' : 'text-gray-400'}`}>{style.name.split('(')[1].replace(')', '')} 스타일</p>

                </button>

              );

            })}

          </div>

        </section>


        {/* Action Button */}

        <div className="flex flex-col items-center gap-4">

          {errorMsg && <div className="text-red-500 bg-red-50 px-4 py-2 rounded-lg text-sm font-medium animate-pulse text-center">{errorMsg}</div>}

          <button onClick={generateImages} disabled={isGenerating || !uploadedImage || selectedStyles.length === 0}

            className={`w-full md:w-auto px-8 py-4 rounded-xl font-bold text-lg shadow-lg flex items-center justify-center gap-2 transition-all ${isGenerating ? 'bg-gray-200 text-gray-500 cursor-not-allowed' : 'bg-gradient-to-r from-indigo-600 to-purple-600 text-white hover:shadow-xl hover:scale-[1.02]'}`}>

            {isGenerating ? <FunLoader /> : <> <Sparkles className="fill-current" /> 선택한 4가지 스타일로 생성하기 </>}

          </button>

        </div>


        {/* Results */}

        {generatedImages.length > 0 && (

          <section className="animate-in fade-in slide-in-from-bottom-4 duration-500">

            <div className="flex items-center justify-between mb-6">

                <h2 className="text-lg font-bold text-gray-800 flex items-center gap-2"><ImageIcon size={20} className="text-indigo-600"/> 생성 결과 ({generatedImages[0].orientation === 'landscape' ? '가로형' : '세로형'} A6)</h2>

                <button onClick={handleDownloadAll} className="flex items-center gap-2 px-4 py-2 bg-indigo-600 hover:bg-indigo-700 text-white rounded-lg text-sm font-bold shadow-md transition-all hover:scale-105"><Download size={16} /> 전체 저장</button>

            </div>

            <div className="grid grid-cols-2 md:grid-cols-4 gap-6">

              {generatedImages.map((img, idx) => (

                <div key={idx} className="group relative bg-white rounded-xl shadow-md overflow-hidden border border-gray-100 hover:shadow-xl transition-shadow">

                  <div className={`relative w-full bg-gray-100 overflow-hidden cursor-zoom-in ${img.orientation === 'landscape' ? 'pb-[70.7%]' : 'pb-[141.4%]'}`} onClick={() => setPreviewImage(img)}>

                    <img src={img.imageData} alt={img.styleName} className="absolute inset-0 w-full h-full object-cover transition-transform duration-500 group-hover:scale-110"/>

                    <div className="absolute inset-0 bg-black/0 group-hover:bg-black/20 transition-colors flex items-center justify-center opacity-0 group-hover:opacity-100"><Maximize2 className="text-white drop-shadow-md" size={32}/></div>

                  </div>

                  <div className="p-3 border-t border-gray-100">

                    <p className="text-xs font-bold text-gray-700 mb-2 truncate flex justify-between"><span>{img.styleName}</span></p>

                    <button onClick={(e) => { e.stopPropagation(); const timestamp = getTimestamp(); handleDownload(img.imageData, `${timestamp}_AI-Art-${img.styleId}.png`); }} className="w-full flex items-center justify-center gap-1.5 bg-indigo-50 hover:bg-indigo-100 text-indigo-700 py-2 rounded-lg text-xs font-semibold transition-colors"><Download size={14} /> 다운로드</button>

                  </div>

                </div>

              ))}

            </div>

          </section>

        )}

      </main>


      {/* Lightbox */}

      {previewImage && (

        <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/95 backdrop-blur-md p-4 animate-in fade-in duration-300" onClick={() => setPreviewImage(null)}>

          <button onClick={() => setPreviewImage(null)} className="absolute top-6 right-6 text-white/70 hover:text-white bg-white/10 hover:bg-white/20 p-2 rounded-full transition-all z-50"><X size={32} /></button>

          

          {generatedImages.length > 1 && (

            <>

              <button onClick={handlePrevPreview} className="absolute left-4 top-1/2 -translate-y-1/2 text-white/70 hover:text-white bg-white/10 hover:bg-white/20 p-3 rounded-full transition-all z-50 backdrop-blur-md hover:scale-110"><ChevronLeft size={48} /></button>

              <button onClick={handleNextPreview} className="absolute right-4 top-1/2 -translate-y-1/2 text-white/70 hover:text-white bg-white/10 hover:bg-white/20 p-3 rounded-full transition-all z-50 backdrop-blur-md hover:scale-110"><ChevronRight size={48} /></button>

            </>

          )}


          <div className="relative w-full max-w-5xl h-full max-h-[95vh] flex flex-col items-center justify-center" onClick={(e) => e.stopPropagation()}>

             <div className="relative w-full h-full flex items-center justify-center">

                <img src={previewImage.imageData} alt="Full Preview" onClick={() => setPreviewImage(null)} className="max-w-full max-h-full rounded-lg shadow-2xl border-2 border-white/20 object-contain cursor-zoom-out"/>

             </div>

             <div className="absolute bottom-8 flex gap-4 animate-in slide-in-from-bottom-4 duration-500">

                <span className="text-white font-medium text-lg bg-black/60 px-6 py-3 rounded-full backdrop-blur border border-white/10">{previewImage.styleName}</span>

                <button onClick={() => { const timestamp = getTimestamp(); handleDownload(previewImage.imageData, `${timestamp}_AI-Art-${previewImage.styleId}-HighRes.png`); }} className="bg-white text-black hover:bg-indigo-50 px-8 py-3 rounded-full font-bold flex items-center gap-2 shadow-lg transition-transform hover:scale-105"><Download size={20} /> 저장하기</button>

                <button onClick={() => handlePrint(previewImage.imageData)} className="bg-indigo-600 text-white hover:bg-indigo-700 px-8 py-3 rounded-full font-bold flex items-center gap-2 shadow-lg transition-transform hover:scale-105"><Printer size={20} /> 인쇄하기</button>

             </div>

          </div>

        </div>

      )}

    </div>

  );

}
